交通灯
PortOut EQU 88H        ; 74LS273输出端口地址
                       ; 所有交通灯状态码都通过该端口输出到LED

START:
    MOV AL,36H         ; 初始状态：东西、南北方向全部红灯
    OUT PortOut,AL     ; 输出到74LS273，锁存后驱动交通灯
    MOV CX,4           ; 设置延时次数，4个半秒约等于2秒
    CALL DELAY_HALF_S  ; 调用延时子程序，让全红状态保持一段时间

TRAFFIC_LOOP:
    MOV AL,33H         ; 状态1：南北绿灯亮，东西红灯亮
    OUT PortOut,AL     ; 输出南北通行状态
    MOV CX,10          ; 10个半秒，约5秒
    CALL DELAY_HALF_S  ; 保持南北绿灯通行

    MOV BP,3           ; 设置南北绿灯闪烁3次

FLASH_NS:
    MOV AL,37H         ; 南北绿灯灭，东西红灯保持
    OUT PortOut,AL     ; 输出绿灯灭状态
    MOV CX,1           ; 延时0.5秒
    CALL DELAY_HALF_S

    MOV AL,33H         ; 南北绿灯重新亮
    OUT PortOut,AL     ; 输出绿灯亮状态
    MOV CX,1           ; 延时0.5秒
    CALL DELAY_HALF_S

    DEC BP             ; 闪烁次数减1
    JNZ FLASH_NS       ; BP不为0，继续闪烁

    MOV AL,35H         ; 南北黄灯亮，东西红灯亮
    OUT PortOut,AL     ; 输出南北黄灯状态
    MOV CX,4           ; 延时约2秒
    CALL DELAY_HALF_S

    MOV AL,1EH         ; 状态3：东西绿灯亮，南北红灯亮
    OUT PortOut,AL     ; 输出东西通行状态
    MOV CX,10          ; 延时约5秒
    CALL DELAY_HALF_S

    MOV BP,3           ; 设置东西绿灯闪烁3次

FLASH_EW:
    MOV AL,3EH         ; 东西绿灯灭，南北红灯保持
    OUT PortOut,AL     ; 输出东西绿灯灭状态
    MOV CX,1           ; 延时0.5秒
    CALL DELAY_HALF_S

    MOV AL,1EH         ; 东西绿灯重新亮
    OUT PortOut,AL     ; 输出东西绿灯亮状态
    MOV CX,1           ; 延时0.5秒
    CALL DELAY_HALF_S

    DEC BP             ; 闪烁次数减1
    JNZ FLASH_EW       ; BP不为0，继续闪烁

    MOV AL,2EH         ; 东西黄灯亮，南北红灯亮
    OUT PortOut,AL     ; 输出东西黄灯状态
    MOV CX,4           ; 延时约2秒
    CALL DELAY_HALF_S

    JMP TRAFFIC_LOOP   ; 一个完整交通灯周期结束，重新循环

延时子程序核心代码
DELAY_HALF_S PROC
    PUSH CX            ; 保护CX，因为主程序用CX表示延时次数
    PUSH DX            ; 保护DX，防止延时过程中破坏原值
    PUSH BX            ; 保护BX，防止延时过程中破坏原值

D_OUTER:
    MOV DX,20          ; 中层循环次数

D_MIDDLE:
    MOV BX,1000        ; 内层循环次数

D_INNER:
    DEC BX             ; BX减1
    JNZ D_INNER        ; BX不为0，继续空转

    DEC DX             ; DX减1
    JNZ D_MIDDLE       ; DX不为0，继续中层循环

    LOOP D_OUTER       ; CX减1，CX不为0则继续外层循环

    POP BX             ; 恢复BX
    POP DX             ; 恢复DX
    POP CX             ; 恢复CX
    RET                ; 返回主程序
DELAY_HALF_S ENDP

外部 SRAM 映射到 90000H～9FFFFH 并写入数据
MemSeg EQU 9000H       ; 定义外部SRAM的段地址
                       ; 9000H × 16 + 0000H = 90000H
                       ; 所以 ES:0000H 对应物理地址90000H

Start:
    MOV AX,MemSeg      ; AX = 9000H
                       ; 8086不能直接把立即数送入ES，所以先送入AX

    MOV ES,AX          ; ES = 9000H
                       ; 后续通过 ES:[DI] 访问90000H开始的外部SRAM

    XOR DI,DI          ; DI = 0000H
                       ; 偏移地址从0000H开始
                       ; 因此第一个访问地址是 ES:0000H，即90000H

    XOR BL,BL          ; BL = 00H
                       ; BL保存当前要写入的数据
                       ; 第一组写入00H，第二组写入01H，依次递增

PageLoop:
    MOV CX,0100H       ; CX = 256
                       ; 每一种数据连续写入256个字节
                       ; 例如00H写满90000H～900FFH

ByteLoop:
    MOV AL,BL          ; 把当前要写入的数据送入AL
                       ; 因为内存写字节时，数据来自AL

    MOV BYTE PTR ES:[DI],AL
                       ; 向 ES:DI 指向的内存单元写入一个字节
                       ; BYTE PTR 表示写入的是一个字节
                       ; 物理地址 = ES × 16 + DI

    INC DI             ; DI加1，指向下一个字节地址
                       ; 例如 0000H → 0001H → 0002H

    LOOP ByteLoop      ; CX减1，若CX不为0则继续写
                       ; 循环256次，完成一个256字节块

    INC BL             ; 写入数据加1
                       ; 00H → 01H → 02H → ... → FFH

    JNZ PageLoop       ; 如果BL加1后不是00H，继续下一组
                       ; 当BL从FFH加1溢出回00H时，说明256组写完

Stop:
    JMP Stop           ; 程序停在这里
                       ; 方便在Proteus Debug中观察U10、U11内容


数码管显示 0、1、2、3
L1:
        MOV AL,0FEH        ; 11111110B，选中第1位数码管
                           ; 位选通常是低电平有效，所以最低位为0表示选中第1位
        OUT PortA,AL       ; 把位选信号送到A口

        MOV AL,SEGTAB      ; 取数字0的段码，即3FH
        OUT PortB,AL       ; B口输出段码，第1位显示0
        CALL DELAY         ; 短延时，让第1位保持一小段时间

        MOV AL,0FDH        ; 11111101B，选中第2位数码管
        OUT PortA,AL
        MOV AL,SEGTAB+1    ; 取数字1的段码
        OUT PortB,AL       ; 第2位显示1
        CALL DELAY

        MOV AL,0FBH        ; 11111011B，选中第3位数码管
        OUT PortA,AL
        MOV AL,SEGTAB+2    ; 取数字2的段码
        OUT PortB,AL       ; 第3位显示2
        CALL DELAY

        MOV AL,0F7H        ; 11110111B，选中第4位数码管
        OUT PortA,AL
        MOV AL,SEGTAB+3    ; 取数字3的段码
        OUT PortB,AL       ; 第4位显示3
        CALL DELAY

        JMP L1             ; 不断重复扫描

PC 口读取 8 位开关输入，高 4 位以二进制形式显示在 LED 上，低 4 位转换成十六进制字符后显示在第一位数码管上。

核心代码整理与详细注释
        MOV AL,10001001B
        OUT CtrlPT,AL

控制字 10001001B 的含义是：8255 工作在方式 0，PA 输出，PB 输出，PC 输入。因为本任务要从 PC 口读取拨位开关，所以 PC 口必须设置为输入。

L1:
        IN AL,PortC             ; 从PC口读取8位开关状态
                                ; 读入后 AL = 开关状态，例如 01100101B

        MOV AH,AL               ; 保存原始开关值
                                ; 因为后面要分别取高4位和低4位
                                ; 所以先把完整数据备份到AH

这里 AH 用作“备份寄存器”。因为 AL 后面会被 AND 修改，如果不备份，原始开关值就丢了。

        MOV AL,AH               ; 取回原始开关值
        AND AL,0F0H             ; 保留高4位，清零低4位
                                ; 例如 01100101B AND 11110000B = 01100000B

        OR AL,00001110B         ; 低4位置为1110，选中第1位数码管
                                ; 高4位仍保留，用于LED显示

        OUT PortA,AL            ; 输出到A口
                                ; A口高4位控制LED
                                ; A口低4位控制数码管位选

这几句同时完成两件事：

AL高4位：来自开关高4位 → 显示到LED
AL低4位：设置为1110 → 选中第一位数码管

所以 PortA 在这里承担了两个功能：高 4 位给 LED，低 4 位做数码管位选。

        MOV AL,AH               ; 再次取回原始开关值
        AND AL,0FH              ; 只保留低4位
                                ; 例如 01100101B AND 00001111B = 00000101B

        MOV BX,OFFSET SEGTAB    ; BX指向段码表首地址
        XLAT                    ; 查表
                                ; XLAT功能：AL = DS:[BX + AL]
                                ; 如果AL=05H，就取SEGTAB第5项，即数字5的段码

        OUT PortB,AL            ; 把查到的七段码输出到B口
                                ; 第一位数码管显示开关低4位对应的十六进制数

        CALL DELAY              ; 短延时，保持显示
        JMP L1                  ; 循环读取开关并刷新
代码理解

这一段最核心的是两个操作：

AND AL,0F0H     ; 取高4位
AND AL,0FH      ; 取低4位

高 4 位直接送 LED，因为 LED 可以直接显示二进制状态。
低 4 位不能直接送数码管，因为数码管需要的是七段码，所以要用 XLAT 查 SEGTAB 表。

可以这样理解：

开关输入 8 位：ABCDEFGH

高 4 位 ABCD → LED 二进制显示
低 4 位 EFGH → 查段码表 → 数码管显示 0~F
